cmake_minimum_required(VERSION 3.21)
project(wallfolio-gui VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Qt6 6.4 REQUIRED COMPONENTS Quick QuickControls2 Network)
qt_standard_project_setup(REQUIRES 6.4)
set(CMAKE_AUTOGEN_PARALLEL 1)
qt_add_executable(wallfolio-gui src/main.cpp src/Client.cpp src/Client.h)
qt_add_qml_module(wallfolio-gui URI Wallfolio VERSION 1.0 RESOURCE_PREFIX / QML_FILES qml/Main.qml)
target_link_libraries(wallfolio-gui PRIVATE Qt6::Quick Qt6::QuickControls2 Qt6::Network)
install(TARGETS wallfolio-gui RUNTIME DESTINATION bin)
